cmake_minimum_required(VERSION 3.22.1)
project(maxbot_native C)
set(CMAKE_C_STANDARD 99)
add_library(maxbot_image SHARED image_jni.c)
target_link_libraries(maxbot_image m)
option(MAXBOT_ENABLE_MPC "Build optional generated MPC solver" OFF)
if(MAXBOT_ENABLE_MPC AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/osqp_generated/workspace.c")
  file(GLOB OSQP_SOURCE "osqp_generated/src/*.c")
  add_library(maxbot_mpc SHARED mpc_jni.c osqp_generated/workspace.c ${OSQP_SOURCE})
  target_include_directories(maxbot_mpc PRIVATE osqp_generated osqp_generated/inc/public osqp_generated/inc/private)
  target_link_libraries(maxbot_mpc m)
endif()
